% latex/conventions.tex — shared preamble and macros for the quant textbook.

% ===== Page layout (match NLP_NOTES.pdf reference: ~1in margins) =====
\setlrmarginsandblock{1in}{1in}{*}
\setulmarginsandblock{1in}{1in}{*}
\checkandfixthelayout

% ===== Encoding & typography =====
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[protrusion=true,expansion=true,final]{microtype}

% ===== Line-breaking slack =====
% Absorb minor overfull hboxes silently by allowing up to 3em of extra glue
% per paragraph as a last resort before breaking a line. This is the standard
% remedy for stubborn overfulls in LaTeX books with narrow columns, long
% technical terms, and inline \texttt{code_identifiers_with_underscores}.
\setlength{\emergencystretch}{3em}
\tolerance=2000
\hbadness=2000

% ===== URL and path line-breaking =====
% Allow breaking long URLs and file paths at any character. Without this,
% bibliography entries with long DOIs/URLs and inline \texttt{research/papers/...}
% paths produce 100+pt overfulls in the bibliography.
\usepackage{xurl}
% xurl's default is to break between any pair of characters. Don't redefine
% \UrlBreaks here — xurl already does the optimal thing.

% Apply \sloppy globally inside the bibliography so long URLs and DOIs never
% create overfull hboxes that escape the text column. This is standard practice
% for books with extensive reference lists.
\usepackage{etoolbox}
\AtBeginEnvironment{thebibliography}{\sloppy}

% ===== Math =====
\usepackage{amsmath, amssymb, amsthm, mathtools}
\usepackage{bm}

% ===== Graphics & figures =====
\usepackage{graphicx}

% Float placement parameters. The memoir defaults are conservative; these
% make figures/tables less likely to be deferred to their own float page,
% which gives a better flow through a math/derivation-heavy book.
\renewcommand{\topfraction}{0.9}         % up to 90% of top of text page is floats
\renewcommand{\bottomfraction}{0.8}      % up to 80% of bottom
\renewcommand{\textfraction}{0.07}       % min 7% of page must be text
\renewcommand{\floatpagefraction}{0.7}   % a float page must be >= 70% floats
\setcounter{topnumber}{3}                % up to 3 floats at top of page
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{5}
% Make default float specifier [tbp] for any float without explicit placement
% (not applied retroactively but useful for consistency).
\usepackage{placeins}                    % provides \FloatBarrier for sections
\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{arrows.meta, positioning, calc, shapes.geometric, decorations.pathreplacing}

% ===== Tables =====
