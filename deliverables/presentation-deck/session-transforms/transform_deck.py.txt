def render_equation_svg(latex: str, *, color: str, fontsize: float = 26.0) -> str:
    """Verbatim copy of generate.py's helper: LaTeX -> inline SVG paths."""
    import matplotlib

    matplotlib.use("Agg")
    matplotlib.rcParams["svg.hashsalt"] = "volforecast-deck"
    from matplotlib.figure import Figure

    fig = Figure(figsize=(0.01, 0.01))
    fig.text(0, 0, f"${latex}$", fontsize=fontsize, color=color,
             math_fontfamily="cm")
    buf = io.BytesIO()
    fig.savefig(buf, format="svg", bbox_inches="tight", pad_inches=0.03,
                transparent=True)
    svg = buf.getvalue().decode("utf-8")
    svg = svg[svg.index("<svg"):]
    svg = re.sub(r"<metadata>.*?</metadata>", "", svg, flags=re.S)
    return re.sub(r"<!--.*?-->", "", svg, flags=re.S)


def render_first_that_works(candidates, **kw):
    last = None
    for latex in candidates:
        try:
            return latex, render_equation_svg(latex, **kw)
        except Exception as e:  # mathtext parse errors
            last = e
    raise last
