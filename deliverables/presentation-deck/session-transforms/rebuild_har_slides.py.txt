"""Rebuild the HAR build slides: stepped equation reveal, Layer 0 table removal,
IV slide tenor diagram.

- Spine slide: five .eq-state blocks (persistence -> +week -> +month -> +IV ->
  assembled w/ labels), matplotlib-mathtext rendered, new term amber per state.
- Deck JS: ArrowRight/Left step through eq-states before changing slides.
- Windows slide: Layer 0 six-column panel removed, diagram recentred, caption cut.
- IV slide: prose tightened + tenor-matching diagram.
"""
import io
import re
from pathlib import Path

import matplotlib

matplotlib.use("Agg")
matplotlib.rcParams["svg.hashsalt"] = "volforecast-deck"
from matplotlib.backends.backend_agg import FigureCanvasAgg
from matplotlib.figure import Figure

DECK = Path(r"C:\Users\RyanPC\Documents\Projects\ML-GS\ml-vol-estimator\workspace\presentation\presentation-latest.html")

BODY, AMBER, MUTED = "#c7cdd6", "#e8b339", "#7d8896"
FS = 26.0
FIG_W_IN, FIG_H_IN, DPI = 40.0, 4.0, 100.0

SEG_LHS = r"\log RV_{t+1} \;=\; \beta_0"
SEG_D = r"\,+\, \beta_d \,\log RV_t"
SEG_W = r"\,+\, \beta_w \,\log\left[\frac{1}{5}\sum_{i=0}^{4} RV_{t-i}\right]"
SEG_M = r"\,+\, \beta_m \,\log\left[\frac{1}{22}\sum_{i=0}^{21} RV_{t-i}\right]"
SEG_IV = r"\,+\, \beta_{iv} \,\log IV_t^{\mathrm{0DTE}}"
SEG_EPS = r"\,+\, \varepsilon_{t+1}"


def render_state(segments, labels=None):
    """segments: list of (latex, color). labels: {seg_index: text} under-centers."""
    fig = Figure(figsize=(FIG_W_IN, FIG_H_IN), dpi=DPI)
    canvas = FigureCanvasAgg(fig)
    x_px, y_frac = 40.0, 0.55
    centers = {}
    for k, (seg, color) in enumerate(segments):
        t = fig.text(x_px / (FIG_W_IN * DPI), y_frac, f"${seg}$",
                     fontsize=FS, color=color, math_fontfamily="cm")
        canvas.draw()
        ext = t.get_window_extent(renderer=canvas.get_renderer())
        centers[k] = (ext.x0 + ext.x1) / 2.0
        x_px = ext.x1
    if labels:
        for k, text in labels.items():
            fig.text(centers[k] / (FIG_W_IN * DPI), y_frac - 0.175, text,
                     fontsize=12.5, color=MUTED, ha="center", fontname="Segoe UI")
    buf = io.BytesIO()
    fig.savefig(buf, format="svg", bbox_inches="tight", pad_inches=0.03,
                transparent=True)
    svg = buf.getvalue().decode("utf-8")
    svg = svg[svg.index("<svg"):]
    svg = re.sub(r"<metadata>.*?</metadata>", "", svg, flags=re.S)
    return re.sub(r"<!--.*?-->", "", svg, flags=re.S)


def namespace_ids(svg, prefix):
    for i in sorted(set(re.findall(r'id="([^"]+)"', svg)), key=len, reverse=True):
        svg = svg.replace(f'id="{i}"', f'id="{prefix}{i}"')
        svg = svg.replace(f'href="#{i}"', f'href="#{prefix}{i}"')
        svg = svg.replace(f'url(#{i})', f'url(#{prefix}{i})')
    return svg


def svg_width_pt(svg):
    return float(re.search(r'width="([\d.]+)pt"', svg).group(1))


states = [
    ([(SEG_LHS, BODY), (SEG_D, AMBER), (SEG_EPS, BODY)], None,
     "Volatility clusters, so start from persistence: tomorrow&rsquo;s variance, regressed on today&rsquo;s."),
    ([(SEG_LHS, BODY), (SEG_D, BODY), (SEG_W, AMBER), (SEG_EPS, BODY)], None,
     "One day is a noisy sample. The five-day average steadies it."),
    ([(SEG_LHS, BODY), (SEG_D, BODY), (SEG_W, BODY), (SEG_M, AMBER), (SEG_EPS, BODY)], None,
     "The month remembers the regime."),
    ([(SEG_LHS, BODY), (SEG_D, BODY), (SEG_W, BODY), (SEG_M, BODY), (SEG_IV, AMBER),
      (SEG_EPS, BODY)], None,
     "Three windows of the past, then one term that looks forward: same-day implied vol, tenor-matched."),
    ([(SEG_LHS, BODY), (SEG_D, BODY), (SEG_W, BODY), (SEG_M, BODY), (SEG_IV, BODY),

]

for k, (segs, labels, gloss) in enumerate(states):
    svg = namespace_ids(render_state(segs, labels), f"hs{k}-")
    rendered.append((svg, gloss))

state_divs = []

    svg = svg.replace("<svg ", f'<svg style="width:{wpx}px;height:auto;" ', 1)
    gloss_html = (f'<p class="dim" style="font-size:16px; margin:12px 0 0;">{gloss}</p>'
                  if gloss else "")
    state_divs.append(
        f'<div class="eq-state"><div class="equation" data-eq="spine">{svg}</div>'
        f'{gloss_html}</div>')

               + "".join(state_divs) + "</div>")

               '<svg viewBox="0 0 620 300" style="width:700px;height:339px;'

html = html[:ws] + new_windows + html[we:]

old_caption = ("<p class=\"dim\">Yesterday alone is seventy-eight numbers, and it overreacts. "

    (106, "5-day forecast", "1-week IV", False),
    (166, "22-day forecast", "1-month IV", False),

old_show = """function show(n) {
  idx = Math.max(0, Math.min(slides.length - 1, n));
  slides.forEach((s, k) => s.classList.toggle('active', k === idx));
  document.getElementById('counter').textContent = (idx + 1) + ' / ' + slides.length;
}"""
new_show = """let eqStep = 0;
function eqStates() {
  return slides[idx] ? [...slides[idx].querySelectorAll('.eq-state')] : [];
}
function renderEq() {
  const st = eqStates();
  st.forEach((el, k) => {
    el.style.display = k === Math.min(eqStep, st.length - 1) ? 'block' : 'none';
  });
  slides[idx].querySelectorAll('.eq-final').forEach(el => {
    el.style.visibility = (st.length && eqStep >= st.length - 1) ? 'visible' : 'hidden';
  });
}
function show(n) {
  idx = Math.max(0, Math.min(slides.length - 1, n));
  eqStep = 0;
  slides.forEach((s, k) => s.classList.toggle('active', k === idx));
  document.getElementById('counter').textContent = (idx + 1) + ' / ' + slides.length;
  renderEq();
}"""

old_keys = """  if (e.key === 'ArrowRight' || e.key === ' ' || e.key === 'PageDown') show(idx + 1);
  else if (e.key === 'ArrowLeft' || e.key === 'PageUp') show(idx - 1);"""
new_keys = """  if (e.key === 'ArrowRight' || e.key === ' ' || e.key === 'PageDown') {
    const st = eqStates();
    if (st.length && eqStep < st.length - 1) { eqStep += 1; renderEq(); }
    else show(idx + 1);
  }
  else if (e.key === 'ArrowLeft' || e.key === 'PageUp') {
    const st = eqStates();
    if (st.length && eqStep > 0) { eqStep -= 1; renderEq(); }
    else show(idx - 1);
  }"""
