"""Move layer-1 material to a dedicated catalogue slide; slim slide 6 to the
BPV-enhanced-target story with an honest squared-vs-products graph.

Slide 6 keeps: motivation line, BPV equation + caption, NEW three-row graph
(returns / squared / neighbour products, shared scale, scale-break on the
jump's square) + RV-vs-BPV sum rail.
New layer-1 slide gets: the two-splits panel (extracted), jump + semivariance
equations, chips with real column names, arrow-stepped in 3 beats.
"""
import math
import re
from pathlib import Path

DECK = Path(r"C:\Users\RyanPC\Documents\Projects\ML-GS\ml-vol-estimator\workspace\presentation\presentation-latest.html")

INK, GREY, MUT, HAIR, AMBER, GREEN, RED = (
    "#e8e4da", "#7d8896", "#9aa5b1", "#1f2a37", "#e8b339", "#4cc38a", "#e05252")
SANS = "'Segoe UI', Verdana, sans-serif"
F = f'font-family="{SANS}"'

html = DECK.read_text(encoding="utf-8")

# ---------------------------------------------------------------------------
# 1. carve up slide 6
# ---------------------------------------------------------------------------
k6 = html.index('a sharper target')
s6 = html.rindex('<section', 0, k6)
e6 = html.index('</section>', k6) + len('</section>')
sec = html[s6:e6]

# old big diagram svg
d0 = sec.index('<div class="diagram" data-diagram="layer1">')
d1 = sec.index('</svg></div>', d0) + len('</svg></div>')
old_svg = sec[d0:d1]

defs_m = re.search(r'<defs>.*?</defs>', old_svg, re.S)
assert defs_m, "defs not found in layer1 svg"
right_m = old_svg.index('<text x="660" y="26"')
right_end = old_svg.rindex('</svg>')
splits_svg = ('<svg viewBox="640 0 540 266" style="width:500px;height:246px;display:block;">'
              + defs_m.group(0) + old_svg[right_m:right_end] + '</svg>')

# equation row children
r0 = sec.index("<div style=\"display:flex; gap:32px; align-items:flex-start; margin-top:16px;\">")
row = sec[r0:sec.rindex('</section>')].rstrip()
assert row.endswith('</div>'), f"row tail: ...{row[-60:]!r}"
c1_end = row.index('jump-robust: a violent bar has no violent neighbour</div></div>') + \
    len('jump-robust: a violent bar has no violent neighbour</div></div>')
bpv_child = row[row.index('>') + 1:c1_end]
c2_start = c1_end
c2_end = row.index('the jump, kept only when the test is confident</div></div>', c2_start) + \
    len('the jump, kept only when the test is confident</div></div>')
jump_child = row[c2_start:c2_end]
semi_block = row[c2_end:-len('</div>')].strip()

# ---------------------------------------------------------------------------
# 2. new slide-6 graph: returns / squared / products, shared scale
# ---------------------------------------------------------------------------
mag = [7, 4, 9, 5, 11, 3, 8, 12, 4, 6, 10, 5, 7, 13, 6, 9, 3, 46, 2, 8, 5, 11, 4, 7, 9, 5, 12, 6]
sign = [1 if (i * 5) % 3 else -1 for i in range(len(mag))]
sign[17] = -1
J = 17
SC = 0.21           # px per squared unit (rows 2 and 3 share it)
CAP = 74            # tower cap with scale break
x0, bw, gp = 150, 20, 10


def bars_x(i):
    return x0 + i * (bw + gp)


p = ['<svg viewBox="0 0 1080 312" style="width:1000px;height:289px;display:block;">']
# row 1: returns
yc = 56
p.append(f'<text x="136" y="{yc + 4}" text-anchor="end" fill="{MUT}" font-size="14" {F}>returns</text>')
p.append(f'<line x1="{x0}" y1="{yc}" x2="{bars_x(len(mag))}" y2="{yc}" stroke="{HAIR}" stroke-width="1"/>')
for i, m in enumerate(mag):
    h = min(m, 26) * (1 if i != J else 1)
    hpx = h if i != J else 40
    y = yc - hpx if sign[i] > 0 else yc
    col = RED if i == J else (GREEN if sign[i] > 0 else "#8a5560")
    p.append(f'<rect x="{bars_x(i)}" y="{y}" width="{bw}" height="{hpx}" fill="{col}" opacity="{1 if i == J else 0.55}"/>')
p.append(f'<text x="{bars_x(J) + bw / 2}" y="{yc + 54}" text-anchor="middle" fill="{RED}" font-size="12" {F}>one violent bar</text>')
# row 2: squared
yb2 = 172
p.append(f'<text x="136" y="{yb2 - 8}" text-anchor="end" fill="{MUT}" font-size="14" {F}>squared</text>')
p.append(f'<line x1="{x0}" y1="{yb2}" x2="{bars_x(len(mag))}" y2="{yb2}" stroke="{HAIR}" stroke-width="1"/>')
for i, m in enumerate(mag):
    v = m * m * SC
    if i == J:
        p.append(f'<rect x="{bars_x(i)}" y="{yb2 - CAP}" width="{bw}" height="{CAP}" fill="{RED}"/>')
        p.append(f'<line x1="{bars_x(i) - 5}" y1="{yb2 - 46}" x2="{bars_x(i) + bw + 5}" y2="{yb2 - 54}" stroke="#0c1117" stroke-width="4"/>')
        p.append(f'<line x1="{bars_x(i) - 5}" y1="{yb2 - 40}" x2="{bars_x(i) + bw + 5}" y2="{yb2 - 48}" stroke="#0c1117" stroke-width="4"/>')
    else:
        p.append(f'<rect x="{bars_x(i)}" y="{yb2 - max(v, 1.2):.1f}" width="{bw}" height="{max(v, 1.2):.1f}" fill="#24303f"/>')
p.append(f'<text x="{bars_x(len(mag)) + 14}" y="{yb2 - 40}" fill="{INK}" font-size="14" {F}>what RV sums</text>')
p.append(f'<text x="{bars_x(len(mag)) + 14}" y="{yb2 - 20}" fill="{GREY}" font-size="13" {F}>the jump owns it (off scale)</text>')
# row 3: neighbour products
yb3 = 268
p.append(f'<text x="136" y="{yb3 - 8}" text-anchor="end" fill="{MUT}" font-size="14" {F}>products</text>')
p.append(f'<line x1="{x0}" y1="{yb3}" x2="{bars_x(len(mag))}" y2="{yb3}" stroke="{HAIR}" stroke-width="1"/>')
for i in range(1, len(mag)):
    v = mag[i] * mag[i - 1] * SC
    hot = i == J or i == J + 1
    col = AMBER if hot else "#24303f"
    p.append(f'<rect x="{bars_x(i)}" y="{yb3 - max(v, 1.2):.1f}" width="{bw}" height="{max(v, 1.2):.1f}" fill="{col}"/>')
p.append(f'<text x="{bars_x(len(mag)) + 14}" y="{yb3 - 40}" fill="{INK}" font-size="14" {F}>what BPV sums</text>')
p.append(f'<text x="{bars_x(len(mag)) + 14}" y="{yb3 - 20}" fill="{GREY}" font-size="13" {F}>one factor of the jump, not two</text>')
# sum rail
rvsum = sum(m * m for m in mag) * SC
bpvsum = sum(mag[i] * mag[i - 1] for i in range(1, len(mag))) * SC
rx = 962
rail_h = 118
p.append(f'<text x="{rx + 40}" y="26" text-anchor="middle" fill="{AMBER}" font-size="12" letter-spacing="2" {F}>THE SUMS</text>')
hb = rail_h
p.append(f'<rect x="{rx}" y="{160 - hb}" width="30" height="{hb}" fill="{RED}" opacity="0.75"/>')
p.append(f'<text x="{rx + 15}" y="{176}" text-anchor="middle" fill="{MUT}" font-size="13" {F}>RV</text>')
hb2 = rail_h * bpvsum / rvsum
p.append(f'<rect x="{rx + 50}" y="{160 - hb2:.0f}" width="30" height="{hb2:.0f}" fill="{GREEN}" opacity="0.85"/>')
p.append(f'<text x="{rx + 65}" y="{176}" text-anchor="middle" fill="{MUT}" font-size="13" {F}>BPV</text>')
p.append(f'<text x="{rx + 40}" y="206" text-anchor="middle" fill="{GREY}" font-size="12" {F}>same day,</text>')
p.append(f'<text x="{rx + 40}" y="222" text-anchor="middle" fill="{GREY}" font-size="12" {F}>two readings</text>')
p.append('</svg>')
new_graph = ('<div class="diagram" data-diagram="layer1">' + "".join(p) + '</div>')

new_sec = (sec[:sec.index('<h1')]
           + '<h1>A reading one violent bar cannot spoil</h1>\n'
           + '<p>Continuous variance is persistent, so it forecasts. Jumps are close to '
             'independent draws, so they do not. So the empirical RV we train on is upgraded '
             'to the day&rsquo;s bipower reading.</p>\n'
           + new_graph
           + '\n<div style="display:flex; gap:32px; align-items:flex-start; margin-top:14px;">'
           + bpv_child
           + '</div>\n</section>')
html = html[:s6] + new_sec + html[e6:]

# ---------------------------------------------------------------------------
# 3. new layer-1 slide between layer 0 and layer 2
# ---------------------------------------------------------------------------
CHIPS = (
    '<div class="feat-grid">'
    '<div class="feat-chip"><div class="feat-name">Continuous</div>'
    '<div class="feat-code">log_bpv_d</div>'
    '<div class="feat-gloss">the persistent part, and the new target</div></div>'
    '<div class="feat-chip"><div class="feat-name">Jump</div>'
    '<div class="feat-code">log_jump_d</div>'
    '<div class="feat-gloss">kept only when the BNS test is confident</div></div>'
    '<div class="feat-chip"><div class="feat-name">Direction</div>'
    '<div class="feat-code">rs_negative &#183; rs_positive</div>'
    '<div class="feat-gloss">the down half and the up half</div></div>'
    '<div class="feat-chip"><div class="feat-name">Asymmetry</div>'
    '<div class="feat-code">signed_jump</div>'
    '<div class="feat-gloss">RS+ minus RS&#8722;</div></div></div>')

caps = ["The day splits by how variance arrived: continuous, or a gated jump.",
        "And by direction: the down half carries the leverage effect.",
        "Four columns the trees can lean on, and the spine's target is the clean one."]
caps_js = ",".join("'" + c.replace("'", "\\'") + "'" for c in caps)

layer1_slide = f"""<section class="slide">
  <div class="kicker">The features &#183; layer 1</div><div class="layer-tag">in the champion</div>
  <div class="rule"></div>
  <h1>Two ways the same variance arrives</h1>
<div class="diagram" data-step-anim="l1anim" id="anim-l1anim" style="margin:6px 0 0;">
<div style="display:flex; gap:40px; align-items:flex-start;">
  <div>{splits_svg}</div>
  <div style="flex:1;">
    <div class="sfade" data-s="0" style="margin-top:6px;">{jump_child}</div>
    <div class="sfade" data-s="1" style="margin-top:18px;">{semi_block}</div>
  </div>
</div>
<div class="sfade" data-s="2">{CHIPS}</div>
<div class="anim-cap" id="cap-l1anim"></div></div>
<script>(function(){{"use strict";
var root=document.getElementById('anim-l1anim');
var cap=document.getElementById('cap-l1anim');
var caps=[{caps_js}];var prevK=-1;
window.stepAnims=window.stepAnims||{{}};
window.stepAnims['l1anim']={{steps:3,go:function(k){{
animApply(root,k);cap.textContent=caps[Math.min(k,caps.length-1)]||'';prevK=k;}}}};
}})();</script>
</section>
"""

kl2 = html.index('<div class="kicker">The features &#183; layer 2</div>')
sl2 = html.rindex('<section', 0, kl2)
html = html[:sl2] + layer1_slide + html[sl2:]

DECK.write_text(html, encoding="utf-8")
n = html.count('<section class="slide')
print(f"done: slide 6 rebuilt, layer 1 slide inserted; {n} sections now")
