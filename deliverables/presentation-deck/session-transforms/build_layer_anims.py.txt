"""Arrow-stepped diagrams for the layer slides + the claim slide's decision scale.

Mechanism: generalizes the equation-reveal stepping. A slide may contain a
[data-step-anim="key"] root; window.stepAnims[key] = {steps, go(k)}. ArrowRight
advances eqStep through those beats before changing slide, ArrowLeft reverses.
Elements with class "sfade" and data-s="N" fade in once step >= N (cumulative).
"""

INK, GREY, MUT, HAIR, AMBER, GREEN, RED, BODY = (
    "#e8e4da", "#7d8896", "#9aa5b1", "#1f2a37", "#e8b339", "#4cc38a", "#e05252", "#c7cdd6")

# ---------------------------------------------------------------------------
# shared snippets

CSS_ADD = """.sfade { opacity: 0; transform: translateY(8px); transition: opacity .45s ease, transform .45s ease; }
.sfade.on { opacity: 1; transform: translateY(0); }
.anim-cap { font-size: 15px; color: #9aa5b1; margin-top: 8px; min-height: 24px; }
"""

JS_HELPER = """
function animStepEls(root){return [...root.querySelectorAll('.sfade')];}
function animApply(root, k){
  animStepEls(root).forEach(el => {
    el.classList.toggle('on', k >= (+el.getAttribute('data-s') || 0));
  });
}
"""


def wrap(key, steps, inner_svg, cap_lines, extra_js=""):
    """Root div + registration IIFE. cap_lines: list per step (caption text)."""
    caps = ",".join("'" + c.replace("'", "\\'") + "'" for c in cap_lines)
    return (
        f'<div class="diagram" data-step-anim="{key}" id="anim-{key}" '
        f'style="margin:6px 0 0;">{inner_svg}'
        f'<div class="anim-cap" id="cap-{key}"></div></div>'
        f'<script>(function(){{"use strict";'
        f'var root=document.getElementById("anim-{key}");'
        f'var cap=document.getElementById("cap-{key}");'
        f'var caps=[{caps}];var prevK=-1;'
        f'{extra_js}'
        f'window.stepAnims=window.stepAnims||{{}};'
        f'window.stepAnims["{key}"]={{steps:{steps},go:function(k){{'
        f'animApply(root,k);cap.textContent=caps[Math.min(k,caps.length-1)]||"";'
        f'if(typeof onStep==="function")onStep(k,prevK);prevK=k;}}}};'
        f'}})();</script>')


def S(x):  # svg text attr shorthand
    return f'font-family="{SANS}"'

def l0_diagram():

    W, H = 1040, 220
    p = [f'<svg viewBox="0 0 {W} {H}" style="width:980px;height:207px;display:block;">']
