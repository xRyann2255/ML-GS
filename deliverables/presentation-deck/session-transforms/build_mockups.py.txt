SERIF = "Georgia, 'Times New Roman', serif"
SANS = "'Segoe UI', Verdana, sans-serif"


def eq_svg(width, font, annotate):
    """The variance-swap payoff equation, hand-positioned tspans.

    sub level +0.30em of font, sup level -0.45em, chained via cumulative dy.
    """
    sub = round(font * 0.30)
    sup_from_sub = -round(font * 0.75)
    restore = -(sub + sup_from_sub)
    small = round(font * 0.58)
    h = 180 if annotate else 110
    s = []
    s.append(f'<svg viewBox="0 0 {width} {h}" style="width:{width}px;height:{h}px;">')
    s.append(
        f'<text x="60" y="70" font-family="{SERIF}" font-size="{font}" '
        f'fill="#e8e4da" font-style="italic">P&amp;L'
        f'<tspan font-style="normal" dx="16">=</tspan>'
        f'<tspan dx="16">N</tspan>'
        f'<tspan font-style="normal" dx="12">&#183;</tspan>'
        f'<tspan font-style="normal" dx="12">(</tspan>'
        f'<tspan dx="6">&#963;</tspan>'
        f'<tspan font-size="{small}" dy="{sub}">R</tspan>'
        f'<tspan font-size="{small}" dy="{sup_from_sub}">2</tspan>'
        f'<tspan font-style="normal" dy="{restore}" dx="14">&#8722;</tspan>'
        f'<tspan dx="14">K</tspan>'
        f'<tspan font-size="{small}" dy="{sub}">var</tspan>'
        f'<tspan font-size="{small}" dy="{sup_from_sub}">2</tspan>'
        f'<tspan font-style="normal" dy="{restore}" dx="8">)</tspan>'
        f'</text>'
    )
    if annotate:
        s.append('<line x1="368" y1="88" x2="368" y2="118" stroke="#1f2a37" stroke-width="1"/>')
        s.append(f'<text x="356" y="140" text-anchor="end" fill="#7d8896" '
                 f'font-size="14" font-family="{SANS}">the variance the day delivers</text>')
        s.append('<line x1="490" y1="88" x2="490" y2="118" stroke="#1f2a37" stroke-width="1"/>')
        s.append(f'<text x="502" y="140" text-anchor="start" fill="#7d8896" '
                 f'font-size="14" font-family="{SANS}">the strike the strip sold at the open</text>')
    s.append("</svg>")
    return "".join(s)
