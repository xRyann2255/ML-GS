def svg_layer1() -> str:
    """Left: why multiplying by your neighbour survives a jump.
    Right: the two decompositions of one day."""
    p = []
    p.append(
        '<svg viewBox="0 0 1180 266" style="width:1080px;height:244px;">'
        '<defs><marker id="arr-l1" viewBox="0 0 10 10" refX="9" refY="5" '
        'markerWidth="7" markerHeight="7" orient="auto-start-reverse">'
        '<path d="M0,0 L10,5 L0,10 z" fill="#7d8896"/></marker>'

        'patternUnits="userSpaceOnUse"><line x1="0" y1="0" x2="0" y2="7" '
        'stroke="#e8b339" stroke-width="2" opacity="0.55"/></pattern></defs>'
    )

    p.append('<text x="20" y="26" fill="#e8b339" font-size="13" '
             'letter-spacing="3">WHY MULTIPLY BY YOUR NEIGHBOUR</text>')

    # Deterministic 5-min return series with one genuine jump.
    n, jump_at = 28, 17
    rets = []
    for k in range(n):
        v = 13.0 * math.sin(k * 1.7) * math.cos(k * 0.9 + 1.0)
        v = max(-15.0, min(15.0, v))
        if abs(v) < 2.5:
            v = 2.5 if v >= 0 else -2.5
        rets.append(v)
    rets[jump_at] = -46.0

    x0, pitch, w = 122, 17, 9
    top = 82
    p.append(f'<line x1="112" y1="{top}" x2="600" y2="{top}" stroke="#1f2a37" stroke-width="1"/>')
    p.append(f'<text x="102" y="{top + 5}" text-anchor="end" fill="#c7cdd6" '
             f'font-size="14">returns</text>')
    for k, v in enumerate(rets):
        bx = x0 + k * pitch
        if k == jump_at:
            fill, op = "#e05252", "1"
        elif v >= 0:
            fill, op = "#4cc38a", "0.55"
        else:
            fill, op = "#e05252", "0.5"
        y = top - v if v >= 0 else top
        p.append(f'<rect x="{bx}" y="{y:.1f}" width="{w}" height="{abs(v):.1f}" '
                 f'fill="{fill}" opacity="{op}"/>')
    jx = x0 + jump_at * pitch + w / 2
    p.append(f'<text x="{jx:.0f}" y="148" text-anchor="middle" fill="#e05252" '
             f'font-size="13">one violent bar</text>')

    p.append('<line x1="356" y1="156" x2="356" y2="188" stroke="#7d8896" '
             'stroke-width="1.5" marker-end="url(#arr-l1)"/>')
    p.append('<text x="368" y="178" fill="#7d8896" font-size="13">'
             'multiply each |r| by the one before it</text>')

    base2 = 236
    p.append(f'<line x1="112" y1="{base2}" x2="600" y2="{base2}" '
             f'stroke="#1f2a37" stroke-width="1"/>')
    p.append(f'<text x="102" y="{base2 + 5}" text-anchor="end" fill="#c7cdd6" '
             f'font-size="14">products</text>')

    # --- right: the two splits ----------------------------------------------
    p.append('<text x="660" y="26" fill="#e8b339" font-size="13" '
             'letter-spacing="3">ONE DAY, TWO SPLITS</text>')

    p.append('<text x="660" y="56" fill="#7d8896" font-size="13">how it arrived</text>')
    p.append('<rect x="660" y="66" width="380" height="36" fill="#1f2a37"/>')
    p.append('<rect x="1040" y="66" width="100" height="36" fill="url(#hatch-l1)"/>')
    p.append('<rect x="660" y="66" width="480" height="36" fill="none" '
             'stroke="#1f2a37" stroke-width="1.5"/>')
    p.append('<text x="674" y="89" fill="#c7cdd6" font-size="14">'

def _eq(parts: list[tuple[str, str]]) -> str:
    """Render an inline equation as SVG tspans.

    parts is a list of (text, kind) with kind in {"n", "sub", "sup"}. dy is
    cumulative in SVG, so each tspan emits the delta from the current shift.
    """
    out, cur = [], 0
    shifts = {"n": 0, "sub": 4, "sup": -6}
    for text, kind in parts:
        target = shifts[kind]
        size = "" if kind == "n" else ' font-size="11"'
        out.append(f'<tspan dy="{target - cur}"{size}>{text}</tspan>')
        cur = target
    if cur:
        out.append(f'<tspan dy="{-cur}"> </tspan>')
    return "".join(out)

def svg_ale() -> str:
    p = ['<svg viewBox="0 0 1180 250" style="width:1080px;height:229px;">']
    pw, gap = 265, 40
    for k, (col, xcap, read) in enumerate(ALE_PANELS):
        g, a = ALE_H1[col]["grid"], ALE_H1[col]["ale"]
        ox = k * (pw + gap)
        px0, px1, py0, py1 = ox + 46, ox + pw, 36, 158     # plot box
        xlo, xhi = min(g), max(g)
        ylo, yhi = min(a), max(a)
        pad = (yhi - ylo) * 0.08 or 1e-9
        ylo, yhi = ylo - pad, yhi + pad
        sx = lambda v: px0 + (v - xlo) / (xhi - xlo) * (px1 - px0)
        sy = lambda v: py1 - (v - ylo) / (yhi - ylo) * (py1 - py0)

        p.append(f'<text x="{ox}" y="20" fill="#e8b339" font-size="12">{col}</text>')
        p.append(f'<line x1="{px0}" y1="{py0}" x2="{px0}" y2="{py1}" '
                 f'stroke="#1f2a37" stroke-width="1"/>')
        zero = sy(0.0)
        if py0 <= zero <= py1:
            p.append(f'<line x1="{px0}" y1="{zero:.1f}" x2="{px1}" y2="{zero:.1f}" '
                     f'stroke="#7d8896" stroke-width="1" stroke-dasharray="3 4" opacity="0.6"/>')
            p.append(f'<text x="{px0 - 6}" y="{zero + 4:.1f}" text-anchor="end" '
                     f'fill="#7d8896" font-size="11">0</text>')
        p.append(f'<text x="{px0 - 6}" y="{py0 + 4}" text-anchor="end" fill="#7d8896" '
                 f'font-size="11">{max(a):+.2f}</text>')
        p.append(f'<text x="{px0 - 6}" y="{py1 + 4}" text-anchor="end" fill="#7d8896" '
                 f'font-size="11">{min(a):+.2f}</text>')
        pts = " ".join(f"{sx(x):.1f},{sy(y):.1f}" for x, y in zip(g, a))
        p.append(f'<polyline points="{pts}" fill="none" stroke="#4cc38a" stroke-width="2"/>')
        p.append(f'<text x="{px0}" y="{py1 + 18}" fill="#7d8896" font-size="11">{xlo:.3g}</text>')
        p.append(f'<text x="{px1}" y="{py1 + 18}" text-anchor="end" fill="#7d8896" '
                 f'font-size="11">{xhi:.3g}</text>')
        p.append(f'<text x="{ox}" y="{py1 + 36}" fill="#9aa5b1" font-size="12">{xcap}</text>')
        for r, line in enumerate(read):
            p.append(f'<text x="{ox}" y="{py1 + 58 + r * 16}" fill="#c7cdd6" '
                     f'font-size="12">{line}</text>')
    p.append("</svg>")
    return "".join(p)

def retarget(html: str) -> str:
    """Replace trial-067 figures inside slides lifted verbatim from the old deck.

    These slides carry hand-built SVGs and the real GSVIVS index series, so they
    are lifted rather than rebuilt; only their numbers are stale.
    """
    # ORDER MATTERS. Every composite phrase must be replaced before the bare
    # numbers inside it, or the bare substitution mutates the phrase and the
    # composite no longer matches. Longest, most specific first.
    subs = [
        # --- composite phrases -------------------------------------------
        # The two Sharpes on the rule slide sit in separate elements, so the
        # pair cannot be matched as one string. Rewrite the "before" here; the
        # bare 2.45 -> 2.57 substitution at the bottom handles the "after".
        ("Annualized Sharpe 2.09", f'Annualized Sharpe {N["sharpe_always"]}'),
        ("with the signal vs 2.09 without",
         f'with the signal vs {N["sharpe_always"]} without'),
        ("Out of sample, purged, five-seeded.",
         f'Out of sample, purged, {N["n_seeds"]}, {N["backtest_days"]} days.'),
        ("peak-to-trough vs always-sell",
         f'peak-to-trough &#183; {N["dd_xgb"]} vs {N["dd_always"]}'),
        ("every headline number is a five-seed mean",
         f'every headline number is a {N["n_seeds"]}'),
        ("One lucky seed looked 6% better than the truth; ", ""),
        ("backtest May 2022 to Jun 2026", "out of sample"),
        ("across all 21 symbols", f'across all {N["n_symbols"]} symbols'),
        # --- slide titles -------------------------------------------------
        ("Walk-forward with a moat, five seeds", "Walk-forward with a moat, three seeds"),
        # --- cross-references: the measurement slide moved 2 -> 3 ---------
        ("the realized-kernel gap from slide 2", "the realized-kernel gap from slide 5"),
        # --- bare numbers, last -------------------------------------------
        ("21 symbols", f'{N["n_symbols"]} symbols'),
        ("2.45", N["sharpe_xgb"]),
        ("33%", N["flat_xgb"]),
        ("41%", N["dd_reduction"]),
    ]
    for old, new in subs:
        html = html.replace(old, new)
    return html


def main() -> None:
    html = SRC.read_text(encoding="utf-8")

    sections = re.findall(r"<section class=\"slide.*?</section>", html, re.S)
    if len(sections) != 11:
        raise SystemExit(f"expected 11 slides in the original, found {len(sections)}")

    head, _, _ = html.partition('<div id="stage">')
    _, _, tail = html.rpartition("</div>\n<button id=\"dashboard-toggle\"")
    tail = '<button id="dashboard-toggle"' + tail

    # Title: lead with the forecast, tease the strategy in one line.
    title = sections[TITLE].replace(
        "A machine-learned realized-variance forecast as a daily trade / stand-aside "
        "signal for the GSVIVS01 index",
        "A machine-learned forecast of SPX realized variance, "
        "and one strategy it pays on",
    )

    # The two strategy slides open the deck, so they keep the original deck's
    # setup kickers ("The product and its problem" / "The claim"). The
    # model-first build used pivot kickers here instead; reinstating them would
    # read as "Now the application" on slide 2, before there is an application.
    product = sections[PRODUCT]
    claim = sections[CLAIM]

    # Act I builds the forecaster one earned term at a time; the equation is a
    # receipt, not a premise. Act II proves it. Act III spends it. Act IV is the
    # linear champion, which is why it appears nowhere before slide 17.
    # Strategy first. The product and the decision we bolt onto it open the
    # deck, so the room knows what the forecast is FOR before a single term of
    # it is built. The build then runs in the same earned order as before,
    # equation last, and the linear champion still appears only in the finale.
    ordered = [
        title,                          # 1  title
        product,                        # 2  GSVIVS01 and its problem
        claim,                          # 3  the 09:10 rule
        slide_naive(),                  # 4  RV_t+1 = RV_t, conceded         [new]
        SLIDE_MEASUREMENT,              # 5  what that number even is
        slide_windows(),                # 6  RV_w, RV_m                      [new]
        slide_iv_tenor(sections[MODEL]),# 7  the forward-looking term        [new]
        slide_equation(sections[MODEL]),# 8  the assembled spine             [new]
        SLIDE_LAYER1,                   # 9  continuous vs jumps
        slide_trees(sections[MODEL]),   # 10 the overlay + the 128 columns   [new]
        sections[VALIDATION],           # 11 walk-forward with a moat
        sections[SHAP],                 # 12 what it learned
        SLIDE_TOP_FEATURES,             # 13 five that don't repeat
        SLIDE_ALE,                      # 14 ALE response curves
        slide_results(),                # 15 where it wins, what it costs    [new]
        slide_finale(),                 # 16 ninety-one percent              [new]
        sections[NUMBERS],              # 17 three numbers
        sections[NEXT],                 # 18 next steps
    ]

    body = retarget("\n".join(s.strip() for s in ordered))

    # The deck's default is `.equation svg { height: 72px; width: auto }`. The
    # HAR line is ~26:1, so a 72px height forces ~1850px of width and it runs
    # off the 1088px content box. Cap this one by width instead.
    head = head.replace(
        ".equation svg { height: 72px; width: auto; }",
        ".equation svg { height: 72px; width: auto; }\n"
        '.equation[data-eq="hariv"] { margin: 12px 0 16px; }\n'
        '.equation[data-eq="hariv"] svg { width: 1020px; height: auto; }',
    )

    out = (
        head.replace(
            "<title>Timing the Variance Seller</title>",
            "<title>Timing the Variance Seller (model first)</title>",
        )
        + '<div id="stage">\n'
        + body
        + "\n</div>\n"
        + tail
    )

    DST.write_text(out, encoding="utf-8")
    n = len(re.findall(r"<section class=\"slide", out))
    print(f"wrote {DST}  ({n} slides, {DST.stat().st_size / 1024:.0f} KB)")


if __name__ == "__main__":
    main()
